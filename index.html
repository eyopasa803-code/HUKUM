# Flask Local Host Sunucu Ayarları
app = Flask(__name__)
DATA_FILE = "localhost_notlar.json"

def load_data():
    if os.path.exists(DATA_FILE):
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            return json.load(f)
    return {"notlar": "", "kodlar": ""}

def save_data(data):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=4)

@app.route("/", methods=["GET", "POST"])
def home():
    data = load_data()
    if request.method == "POST":
        data["notlar"] = request.form.get("notlar", "")
        data["kodlar"] = request.form.get("kodlar", "")
        save_data(data)
    
    html_template = """
    
    
    HÜKÜM Not Panel
    
        
🛡️ HÜKÜM LOCAL HOST NOT PANELİ

        
Çıktığınızda veriler otomatik olarak 'localhost_notlar.json' dosyasına kaydedilir.

        

            
📝 NOTLARINIZ:

            
            
💻 KOD KAYDETME ALANI:

            
            


        

    
    
    """
    return render_template_string(html_template, data=data)

def site_ip_bul():
    hedef = input("\nHedef site adresini girin (Örn: google.com): ").strip()
    if "http" in hedef:
        hedef = urlparse(hedef).netloc
    try:
        ip = socket.gethostbyname(hedef)
        print(f"✅ Hedef Site: {hedef} -> IP Adresi: {ip}")
    except Exception as e:
        print(f"❌ Hata: IP adresi çözümlenemedi. ({e})")

def ip_sorgu():
    ip = input("\nSorgulanacak IP Adresini Girin: ").strip()
    print("🔄 API'den gerçek zamanlı bilgiler çekiliyor...")
    try:
        response = requests.get(f"http://ip-api.com/json/{ip}").json()
        if response.get("status") == "success":
            print(f"\n📍 --- {ip} İÇİN GERÇEK BİLGİLER ---")
            print(f"Ülke     : {response.get('country')} ({response.get('countryCode')})")
            print(f"Bölge/İl : {response.get('regionName')}")
            print(f"Şehir    : {response.get('city')}")
            print(f"Posta K. : {response.get('zip')}")
            print(f"İnternet Sağlayıcı (ISP): {response.get('isp')}")
            print(f"Koordinat: {response.get('lat')}, {response.get('lon')}")
            print(f"Harita   : https://www.google.com/maps/?q={response.get('lat')},{response.get('lon')}")
        else:
            print("❌ Geçersiz IP veya API hatası.")
    except Exception as e:
        print(f"❌ Bağlantı hatası: {e}")

def whois_sorgu():
    domain = input("\nDomain girin (Örn: örnek.com): ").strip()
    try:
        res = requests.get(f"https://rdap.org/domain/{domain}").json()
        print(f"\n🔍 {domain} WHOIS Detayları:")
        if "handle" in res:
            print(f"Kayıt Kimliği: {res.get('handle')}")
            print(f"Durum: {', '.join(res.get('status', []))}")
        else:
            print("❌ Detaylı bilgi bulunamadı veya domain geçersiz.")
    except:
        print("❌ WHOIS sorgusu başarısız oldu.")

def port_tara():
    hedef = input("\nHedef IP veya Site girin: ").strip()
    if "http" in hedef:
        hedef = urlparse(hedef).netloc
    try:
        target_ip = socket.gethostbyname(hedef)
        print(f"⚡ {target_ip} üzerinde popüler portlar taranıyor...")
        populer_portlar = [21, 22, 23, 25, 53, 80, 110, 443, 3306, 8080]
        for port in populer_portlar:
            s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            s.settimeout(1.0)
            result = s.connect_ex((target_ip, port))
            if result == 0:
                print(f"🔓 Port {port}: AÇIK")
            s.close()
        print("✅ Tarama tamamlandı.")
    except:
        print("❌ Bağlantı kurulamadı.")

def dns_sorgu():
    hedef = input("\nHedef domain girin: ").strip()
    print(f"🔍 {hedef} için temel DNS çözümlemesi yapılıyor...")
    try:
        ip = socket.gethostbyname(hedef)
        print(f"A Kaydı (IP): {ip}")
        print(f"Kanonik İsim: {socket.gethostbyaddr(ip)[0]}")
    except:
        print("❌ DNS bilgisi alınamadı.")

def http_header_analiz():
    hedef = input("\nHedef URL girin (http:// veya https:// dahil): ").strip()
    try:
        response = requests.get(hedef, timeout=5)
        print(f"\n🖥️ {hedef} Sunucu Başlık Bilgileri:")
        for key, value in response.headers.items():
            print(f"{key}: {value}")
    except Exception as e:
        print(f"❌ Sunucuya bağlanılamadı: {e}")

def robots_txt_kontrol():
    hedef = input("\nHedef siteyi girin (Örn: https://google.com): ").strip()
    url = f"{hedef.rstrip('/')}/robots.txt"
    try:
        res = requests.get(url, timeout=5)
        if res.status_code == 200:
            print(f"\n📄 Robots.txt İçeriği ({url}):\n")
            print(res.text[:1000])
        else:
            print(f"❌ Robots.txt bulunamadı. Durum kodu: {res.status_code}")
    except:
        print("❌ Siteye erişilemedi.")

def sifre_testi():

def sosyal_medya_hesap_paneli():
    print("\n👥 Çoklu Hesap Giriş İstasyonuna Hoş Geldiniz (1-50 Serbest Seçim)")
    try:
        adet = int(input("Kaç adet hesap tanımlamak istiyorsunuz? (1-50 arası): "))
        if adet < 1 or adet > 50:
            print("⚠️ Lütfen 1 ile 50 arasında bir sayı seçin.")
            return
    except ValueError:
        print("❌ Geçersiz sayı girdiniz.")
        return

    hesaplar = {}
    for i in range(1, adet + 1):
        print(f"\n[Hesap {i}]")
        kullanici = input(f"Kullanıcı Adı veya E-posta (Boş bırakmak için Enter): ").strip()
        if not kullanici:
            print(f"⏩ Hesap {i} es geçildi.")
            continue
        sifre = input(f"Hesap {i} Şifresi: ").strip()
        hesaplar[f"Hesap_{i}"] = {"user": kullanici, "pass": sifre}
    print(f"\n✅ Toplam {len(hesaplar)} hesap yerel otomasyon havuzuna kaydedildi.")
